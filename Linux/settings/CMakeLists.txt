pkg_check_modules(ADW IMPORTED_TARGET libadwaita-1 gtk4 fontconfig)
if(ADW_FOUND)
  add_executable(vikey-settings Settings.cpp Guide.cpp Emoji.cpp ../common/Prefs.cpp)
  target_compile_definitions(vikey-settings PRIVATE VIKEY_VERSION="${VIKEY_VERSION}" VIKEY_DATADIR="${CMAKE_INSTALL_FULL_DATADIR}/vikey")
  target_link_libraries(vikey-settings PRIVATE PkgConfig::ADW)
  pkg_check_modules(GTKX11 IMPORTED_TARGET gtk4-x11 x11)
  if(GTKX11_FOUND)   # the emoji picker places itself in the middle of the screen (X11; Wayland has no window positions)
    target_compile_definitions(vikey-settings PRIVATE VIKEY_HAVE_X11)
    target_link_libraries(vikey-settings PRIVATE PkgConfig::GTKX11)
  endif()
  install(TARGETS vikey-settings DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES vikey.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES ${ROOT}/site/assets/icon.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/512x512/apps RENAME vikey.png)
else()
  message(WARNING "libadwaita not found: vikey-settings not built")
endif()
